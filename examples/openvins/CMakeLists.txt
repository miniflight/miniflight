cmake_minimum_required(VERSION 3.16)
project(miniflight_vio LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
find_package(Eigen3 REQUIRED)
find_package(OpenCV REQUIRED)
find_package(Boost REQUIRED COMPONENTS filesystem thread date_time)
find_package(Ceres REQUIRED)
if(NOT EXISTS "${OPENVINS_SOURCE}/ov_msckf/src/core/VioManager.cpp")
  message(FATAL_ERROR "Set OPENVINS_SOURCE to an OpenVINS checkout")
endif()
file(GLOB_RECURSE sources "${OPENVINS_SOURCE}/ov_core/src/*.cpp" "${OPENVINS_SOURCE}/ov_init/src/*.cpp")
list(FILTER sources EXCLUDE REGEX "/test_[^/]*\\.cpp$|/sim/")
file(GLOB estimator "${OPENVINS_SOURCE}/ov_msckf/src/core/*.cpp" "${OPENVINS_SOURCE}/ov_msckf/src/state/*.cpp" "${OPENVINS_SOURCE}/ov_msckf/src/update/*.cpp")
set(cv_libraries ${OpenCV_LIBRARIES})
list(FILTER cv_libraries INCLUDE REGEX "opencv_(core|imgproc|imgcodecs|video|features|features2d|geometry|calib|calib3d|flann)$")
add_library(openvins_core STATIC ${sources} ${estimator})
target_include_directories(openvins_core PUBLIC ${OpenCV_INCLUDE_DIRS} "${OPENVINS_SOURCE}/ov_core/src" "${OPENVINS_SOURCE}/ov_init/src" "${OPENVINS_SOURCE}/ov_msckf/src")
target_compile_definitions(openvins_core PUBLIC ROS_AVAILABLE=0 ENABLE_ARUCO_TAGS=0)
# OpenVINS relies on transitive includes supplied by older Eigen/OpenCV versions.
target_compile_options(openvins_core PUBLIC "SHELL:-include cassert" "SHELL:-include opencv2/calib3d.hpp")
target_link_libraries(openvins_core PUBLIC Eigen3::Eigen Ceres::ceres Boost::filesystem Boost::thread Boost::date_time ${cv_libraries})
add_library(miniflight_vio SHARED api.cpp)
target_link_libraries(miniflight_vio PRIVATE openvins_core)
